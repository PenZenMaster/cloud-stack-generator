# Enhanced WebsiteGenerationWorker with better error handling


class WebsiteGenerationWorkerFixed(QThread):
    """Enhanced worker thread with comprehensive error handling and logging."""

    progress_updated = pyqtSignal(str)  # Progress message
    generation_completed = pyqtSignal(dict)  # Generation results
    generation_failed = pyqtSignal(str)  # Error message

    def __init__(self, config: ClientConfig, output_dir: str):
        super().__init__()
        self.config = config
        self.output_dir = output_dir

    def run(self):
        """Run website generation with comprehensive error handling."""
        try:
            print(f"🔍 DEBUG: Worker thread started, output dir: {self.output_dir}")
            self.progress_updated.emit("Initializing website generation...")

            # Verify output directory
            output_path = Path(self.output_dir)
            if not output_path.exists():
                print(f"🔍 DEBUG: Creating output directory: {output_path}")
                output_path.mkdir(parents=True, exist_ok=True)

            print("🔍 DEBUG: Creating generator manager")

            # Import and create generator manager
            try:
                from src.core.generators import GeneratorManager

                generator = GeneratorManager()
                print("🔍 DEBUG: Generator manager created successfully")

            except Exception as import_error:
                print(f"🔍 DEBUG: Generator import error in worker: {import_error}")
                self.generation_failed.emit(
                    f"Could not import generator: {str(import_error)}"
                )
                return

            # Connect progress signals if available
            if hasattr(generator, "progress_updated"):
                generator.progress_updated.connect(self.progress_updated.emit)
                print("🔍 DEBUG: Progress signals connected")

            print("🔍 DEBUG: Starting website generation")
            self.progress_updated.emit("Generating website files...")

            # Generate website
            result = generator.generate_website(self.config, self.output_dir)
            print(f"🔍 DEBUG: Generation result: {result}")

            if result and result.get("success", False):
                print("🔍 DEBUG: Generation successful")
                self.generation_completed.emit(result)
            else:
                error_msg = (
                    result.get("error", "Unknown error occurred")
                    if result
                    else "No result returned"
                )
                print(f"🔍 DEBUG: Generation failed: {error_msg}")
                self.generation_failed.emit(error_msg)

        except Exception as e:
            print(f"🔍 DEBUG: Worker thread exception: {e}")
            import traceback

            traceback.print_exc()
            logger.error(f"Website generation worker error: {e}")
            self.generation_failed.emit(f"Generation failed: {str(e)}")
